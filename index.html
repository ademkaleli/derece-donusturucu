📌 Dönüşüm Formülleri:
• Radyan = Derece × (π / 180)
• Grad = Derece × (200 / 180)

☕ Java Kodu (DereceDonusturucu.java)
package derecedonusturucu;

import java.util.Scanner;

public class DereceDonusturucu {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        
        System.out.print("Açı değerini derece cinsinden giriniz: ");
        double derece = scanner.nextDouble();

        double radyan = derece * (Math.PI / 180.0);
        double grad = derece * (200.0 / 180.0);

        System.out.printf("Radyan Karşılığı: %.4f rad
", radyan);
        System.out.printf("Grad Karşılığı  : %.2f grad
", grad);

        scanner.close();
    }
}
